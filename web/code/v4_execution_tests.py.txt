import unittest
from v4_execution import execute
class ExecutionOracles(unittest.TestCase):
    def base(self,**kw):
        x=dict(budget=1000,entry=300,target=300,fx0=1,fx1=1,bps=0,fee=0,fx_bps=0,cash_rate=0,shock=0)
        x.update(kw);return execute(**x)
    def test_integer_cash_conservation(self):
        x=self.base();self.assertEqual(x['units'],3);self.assertEqual(x['cash_usd'],100);self.assertEqual(x['terminal_sgd'],1000)
    def test_flat_fractional_equals_integer(self):
        self.assertAlmostEqual(self.base(integer=False)['terminal_sgd'],1000)
    def test_fixed_costs_both_sides(self):
        x=self.base(fee=1);self.assertEqual(x['cash_usd'],99);self.assertEqual(x['terminal_sgd'],998)
    def test_cash_only_interest(self):
        self.assertAlmostEqual(self.base(cash_rate=.1)['terminal_sgd'],900+100*1.1**3)
    def test_zero_units_no_equity_costs(self):
        x=self.base(entry=2000,fee=100);self.assertEqual(x['units'],0);self.assertEqual(x['terminal_sgd'],1000)
    def test_fx_two_sides(self):
        x=self.base(entry=2000,fx_bps=100);self.assertAlmostEqual(x['terminal_sgd'],1000*.99*.99)
    def test_same_terminal_different_path_risk(self):
        a=self.base(shock=0);b=self.base(shock=-.3);self.assertEqual(a['terminal_sgd'],b['terminal_sgd']);self.assertLess(b['path_max_drawdown'],a['path_max_drawdown'])
    def test_exit_fx_scales_terminal(self):
        self.assertEqual(self.base(fx1=1.2)['terminal_sgd'],1200)
    def test_four_year_path_contains_the_assumed_trough(self):
        r=self.base(entry=100,target=100,years=4,shock=-.3)
        self.assertAlmostEqual(min(p['price']for p in r['path']),70)
        self.assertAlmostEqual(r['path_max_drawdown'],-.3)
    def test_fractional_year_path_contains_the_assumed_trough(self):
        r=self.base(entry=100,target=100,years=3.5,shock=-.4)
        self.assertAlmostEqual(r['path_max_drawdown'],-.4)
    def test_unbounded_horizon_rejected(self):
        with self.assertRaises(ValueError):self.base(years=1e100)
if __name__=='__main__':unittest.main()
